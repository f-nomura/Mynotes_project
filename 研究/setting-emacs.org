(defun my/trash-cleanup (&optional days)
  "XDGゴミ箱 (~/.local/share/Trash) 内の30日以上前のファイルを削除する。
DAYS が指定されていれば、その日数を閾値に使う。"
  (interactive) 
  (let* ((days (or days 30))
         (trash-dir (expand-file-name "~/.local/share/Trash/files"))
         (info-dir  (expand-file-name "~/.local/share/Trash/info"))
         (now (float-time (current-time)))
         (threshold (* 86400 days)))
    (dolist (dir (list trash-dir info-dir))
      (when (file-directory-p dir)
        (dolist (file (directory-files dir t "^[^.]" t))
          (let* ((attrs (file-attributes file))
                 (mtime (float-time (file-attribute-modification-time attrs)))
                 (age (- now mtime)))
            (when (> age threshold)
              (condition-case err
                  (progn
                    (message "✔ Deleting old trash item: %s" file)
                    (if (file-directory-p file)
                        (delete-directory file t)
                      (delete-file file)))
                (error (message "⚠️ Skipped (no permission): %s" file))))))))))

(defun my/wsl-open-with-windows-chrome (path)
  "Open PATH using Windows Chrome via PowerShell.
Works for both /mnt/... and /home/... paths."
  (let* ((win-path
          ;; wslpath -w で Windows のパスに変換
          (string-trim
           (shell-command-to-string
            (format "wslpath -w '%s'" path)))))
    (start-process
     "chrome-wsl" nil
     "/mnt/c/Windows/System32/WindowsPowerShell/v1.0/powershell.exe"
     "-NoProfile" "-Command"
     (format "Start-Process 'C:\\Program Files\\Google\\Chrome\\Application\\chrome.exe' '%s'"
             win-path))))

;; Treemacs のノードを Windows Chrome で開く
(defun my/treemacs-open-in-windows-chrome ()
  "Open the current Treemacs file node in Windows Chrome."
  (interactive)
  (let* ((btn  (treemacs-node-at-point))
         (path (and btn (treemacs-safe-button-get btn :path))))
    (cond
     ;; node がない
     ((not path)
      (message "No node at point."))

     ;; ディレクトリは開かない
     ((not (file-regular-p path))
      (message "This node is not a file: %s" path))

     ;; OK → Windows Chrome で開く
     (t
      (my/wsl-open-with-windows-chrome path)
      (message "Opening in Windows Chrome: %s" path)))))

      (user-error "このバッファはファイルに保存されていません"))
    (file-name-directory (file-truename buffer-file-name)))

  (defun my/processing-sketch-name ()
    "Return the current sketch directory name."
    (file-name-nondirectory
     (directory-file-name (my/processing-sketch-dir))))

  (defun my/processing-main-file-name ()
    "Return current file base name without extension."
    (unless buffer-file-name
      (user-error "このバッファはファイルに保存されていません"))
    (file-name-base buffer-file-name))

  (defun my/processing-validate-sketch ()
    "Validate Processing sketch naming rule."
    (let ((sketch-name (my/processing-sketch-name))
          (main-name   (my/processing-main-file-name)))
      (unless (string= sketch-name main-name)
        (user-error
         "Processing の制約: ディレクトリ名 '%s' とファイル名 '%s.pde' を一致させてください"
         sketch-name main-name))))

  (defun my/processing-save-pde-buffers ()
    "Save all modified .pde buffers in the current sketch directory."
    (let ((sketch-dir (file-truename (my/processing-sketch-dir))))
      (save-some-buffers
       t
       (lambda ()
         (and buffer-file-name
              (string-match-p "\\.pde\\'" buffer-file-name)
              (string-prefix-p sketch-dir (file-truename buffer-file-name)))))))

  (defun my/processing-output-dir (name)
    "Return output directory NAME under current sketch."
    (expand-file-name name (my/processing-sketch-dir)))

  (defun my/processing-action-arg (action)
    "Return CLI flag for ACTION."
    (pcase action
      ('run     "--run")
      ('present "--present")
      ('build   "--build")
      ('export  "--export")
      (_ (user-error "Unknown action: %S" action))))

  (defun my/processing-git-bash-command (action &optional output-dir)
    "Build command string executed inside Git Bash."
    (let* ((sketch-dir (my/processing--to-windows-path
                        (my/processing-sketch-dir)))
           (output-arg
            (when output-dir
              (concat "--output="
                      (shell-quote-argument
                       (my/processing--to-windows-path output-dir)))))
           (processing-command
            (string-join
             (delq nil
                   (list my/processing-cli-command
                         "cli"
                         (concat "--sketch=" (shell-quote-argument sketch-dir))
                         output-arg
                         "--force"
                         (my/processing-action-arg action)))
             " "))
           (java-environment
            (concat "export JAVA_TOOL_OPTIONS="
                    (shell-quote-argument my/processing-java-tool-options))))
      ;; stderr を stdout に統合し、Emacs 側の :coding を一元適用させる
      (concat java-environment " && " processing-command " 2>&1")))

    "Return Processing output buffer."
    (get-buffer-create my/processing-buffer-name))

  (defun my/processing-show-buffer ()
    "Show Processing output buffer."
    (interactive)
    (display-buffer (my/processing-buffer)))

  (defun my/processing--insert-line (text)
    "Insert TEXT at end of Processing buffer."
    (with-current-buffer (my/processing-buffer)
      (let ((inhibit-read-only t))
        (goto-char (point-max))
        (insert text)
        (unless (string-suffix-p "\n" text)
          (insert "\n")))))

  (defun my/processing--process-filter (_proc output)
    "Append OUTPUT to the Processing buffer."
    (with-current-buffer (my/processing-buffer)
      (let ((inhibit-read-only t)
            (moving (= (point) (point-max))))
        (goto-char (point-max))
        (insert output)
        (when moving (goto-char (point-max))))))

  (defun my/processing--process-sentinel (proc event)
    "Handle Processing process PROC state change EVENT."
    (when (memq (process-status proc) '(exit signal))
      (my/processing--insert-line
       (format "Processing %s" (string-trim event)))
      (setq my/processing--process nil)))

  (defun my/processing-stop ()
    "Stop current Processing process."
    (interactive)
    (if (and my/processing--process
             (process-live-p my/processing--process))
        (progn
          (interrupt-process my/processing--process)
          (message "Processing を停止しました"))
      (message "実行中の Processing はありません")))

  (defun my/processing--run-with-action (action &optional output-dir)
    "Run Processing with ACTION, saving buffers first."
    (my/processing-validate-sketch)
    (my/processing-save-pde-buffers)

    (when (and my/processing--process
               (process-live-p my/processing--process))
      (user-error "すでに Processing が実行中です"))

    (let* ((default-directory (my/processing-sketch-dir))
           (buffer (my/processing-buffer))
           (inner-cmd (my/processing-git-bash-command action output-dir))
           ;; make-process 生成前後の既定 coding も固定しておく
           (coding-system-for-read  'utf-8-unix)
           (coding-system-for-write 'utf-8-unix))
      (with-current-buffer buffer
        (let ((inhibit-read-only t))
          (read-only-mode -1)
          (erase-buffer)
          (insert (format "Processing started: %s\n\n" inner-cmd))
          (special-mode)))

      (setq my/processing--process
            (make-process
             :name "processing"
             :buffer buffer
             :command (list my/processing-git-bash-executable "-lc" inner-cmd)
             :coding '(utf-8-unix . utf-8-unix)
             :connection-type 'pipe
             :filter   #'my/processing--process-filter
             :sentinel #'my/processing--process-sentinel))
      ;; :stderr は指定しない（2>&1 で統合済み）

      (display-buffer buffer)))

  ;; ---------------- Interactive commands ----------------

  (defun my/processing-run ()
    "Run current Processing sketch."
    (interactive)
    (my/processing--run-with-action 'run))

  (defun my/processing-present ()
    "Run current Processing sketch in presentation mode."
    (interactive)
    (my/processing--run-with-action 'present))

  (defun my/processing-build ()
    "Build current Processing sketch."
    (interactive)
    (let ((output-dir (my/processing-output-dir "build")))
      (make-directory output-dir t)
      (my/processing--run-with-action 'build output-dir)))

  (defun my/processing-export ()
    "Export current Processing sketch."
    (interactive)
    (let ((output-dir (my/processing-output-dir "export")))
      (make-directory output-dir t)
      (my/processing--run-with-action 'export output-dir)))

        :desc "Processing export"  "e" #'my/processing-export

**** Orderless の設定

[[https://github.com/oantolin/orderless][github/orderless]]

Emacs 全体の補完スタイルを Orderless ベースにする．
Emacs デフォルトでは前方一致検索のみに対応しているところを，より柔軟な検索を可能にする

#+begin_src emacs-lisp :tangle no

#+end_src

#+begin_src emacs-lisp
(use-package! orderless
  :init
  (setq completion-styles '(orderless basic)
        completion-category-defaults nil
        ;; completion-category-overrides '((file (style particul-completion)))
        )
  :config
  (with-eval-after-load 'migemo
    (defun orderless-migemo (component)
      (let ((pattern (downcase (migemo-get-pattern component))))
        (condition-case nil
            (progn (string-match-p pattern "") pattern)
          (invalid-regexp nil))))
    (add-to-list 'orderless-matching-styles 'orderless-migemo))

  (setq orderless-matching-styles 
        '(orderless-literal 
          orderless-prefixes
          orderless-initialism
          orderless-regexp))

)
#+end_src

***** init

- =completion-styles '(orderless basic=: 補完スタイルの優先順位を 1. orderless, 2. emacs にする．
- =completion-category-defaults nil=: Emacs にはファイル名やバッファ名などカテゴリ単位のデフォルト補完設定があるが，orderless を全体に効かせたい場合，これを nil にする．
- =completion-category-overrides '((file (style particul-completion)))=: ファイル補完は orderless ではなく， =particul-completion= を使う．

***** migemo

- =(with-eval-after-load 'migemo=: migemo がロードされてから設定する
- =(let ((pattern (downcase (migemo-get-pattern component))))=: =component= (orderless の検索語句)を =migemo-get-pattern= に入力して日本語の正規表現にする．
   
***** orderless-matching-styles

- orderless-literal: 完全一致
- orderless-prefixes: 前方一致
- orderless-initialism: 頭文字一致(ff -> find-file)
- orderless-regexp: 正規表現でマッチ

***** orderless-style-dispatchers(不使用)

特定の入力文字列に対してどのマッチングを用いるかを制御する． =orderless-affix-dispatch= は=,$,^のような接辞で切り替え．

***** org-edit-special 用 eglot の設定

org-babel の C-c '(org-edit-special)で開かれる分離バッファでも Eglot を動かすための設定
org-edit-special は
- buffer-file-name が nil: LSP はファイル名(または path)が必要
- default-directory が不定: LSP はバッファがどのプロジェクトに属しているかを判断して LSP を起動する．そもそもプロジェクトルートがわからないと起動しない．
などの理由で eldoc が使用できなかったので，擬似的に属するプロジェクトと buffer-file-name を与える．

1. default-directory を元の org ファイルの場所に設定
2. 擬似的な buffer-file-name を与える

  "major-modeに応じて拡張子つきの擬似ファイル名を返す。"
  (concat ".org-src-" (md5 (format "%s:%s" (buffer-name) mode))
          (pcase mode
            ('julia-mode ".jl")
            ('python-mode ".py")
            ('ess-r-mode  ".R")
            ('LaTeX-mode  ".tex")

                                    default-directory))

      (setq-local buffer-offer-save nil)
      (setq-local make-backup-files nil)
      (setq-local auto-save-default nil))

    ;; Eglotを起動・このバッファを管理対象に
    (eglot-ensure)

    ;; ;; CAPFにEglotを最優先で差し込む（他のCAPFと共存可）
    ;; (add-hook 'completion-at-point-functions
    ;;           #'eglot-completion-at-point

    ))

  "WSLのUNIX-PATHをWindowsパスに変換する。"

(defun my/org--powershell (script)
  "PowerShell.exeでSCRIPT を実行し、終了コードを返す。"
  (call-process "powershell.exe" nil nil nil
                "-NoProfile" "-Sta" "-Command"

(defun my/paste-image-to-fig ()
  "クリップボードの画像をfig/に保存し、orgに挿入する。"
  (interactive)

  (unless buffer-file-name
    (user-error "このバッファでは画像を挿入できません"))

  (let* ((base-dir  (file-name-directory buffer-file-name))
         (fig-dir   (expand-file-name "fig/" base-dir))
         (_         (make-directory fig-dir t))
         (name-base (file-name-sans-extension
                     (file-name-nondirectory buffer-file-name)))
         (timestamp (format-time-string "%Y%m%d_%H%M%S"))
         (target    (expand-file-name
                     (format "%s_%s.png" name-base timestamp)
                     fig-dir))
         (win-path  (my/org--as-windows-path target))
         (ps-script (format "(Get-Clipboard -Format image).Save('%s')" win-path)))

    (unless (zerop (my/org--powershell ps-script))
      (user-error "クリップボードに画像がありません"))

    (insert (format "[[%s]]" target))
    (org-display-inline-images)))

#+begin_src emacs-lisp
(defvar my/org-latex-process nil
  "latexmk 非同期プロセス")

(defvar my/org-latex-export-timer 0
  "Counts seconds while latexmk is running.")

(defvar my/org-latex-export-spinner
  '("◯" "◎" "◉" "◎")
  "Spinner icons for latexmk progress.")

(defun my/org-latex-export-monitor ()
  "Monitor latexmk process and show progress in minibuffer."
  (when (and my/org-latex-process
             (process-live-p my/org-latex-process))
    (setq my/org-latex-export-timer
          (1+ my/org-latex-export-timer))
    (let* ((n (length my/org-latex-export-spinner))
           (icon (nth (mod my/org-latex-export-timer n)
                      my/org-latex-export-spinner)))
      (message "%s PDF生成中… %ds"
               icon my/org-latex-export-timer))
    (run-at-time 1 nil #'my/org-latex-export-monitor)))

(defun my/org-latex-export-start-monitor ()
  "Start monitoring latexmk process."
  (setq my/org-latex-export-timer 0)
  (message "↪️ PDF生成開始…")
  (run-at-time 1 nil #'my/org-latex-export-monitor))

(defun my/org-latex-export-async ()
  "Org → LaTeX → latexmk を非同期で実行し、進行状況を message に表示する。"
  (interactive)

  ;; 二重起動防止
  (when my/org-latex-process
    (user-error "すでに PDF 生成中です"))

  ;; Org バッファ確認
  (unless (derived-mode-p 'org-mode)
    (user-error "Org バッファではありません"))

  ;; 保存ファイル確認
  (unless buffer-file-name
    (user-error "この Org ファイルは保存されていません"))

  ;; 念のため保存
  (save-buffer)

  (let* ((org-file buffer-file-name)
         (tex-file (file-name-nondirectory
                    (concat (file-name-sans-extension org-file) ".tex")))
         (default-directory (file-name-directory org-file)))

    ;; Org → LaTeX（同期）
    (message "⚡ Org → LaTeX 変換中…")    
    (org-latex-export-to-latex)

    ;; latexmk 非同期開始
    (setq my/org-latex-process
          (start-process
           "org-latexmk" "*org-latexmk*"
           "latexmk" "-f" tex-file))

    ;; 進行状況表示開始
    (my/org-latex-export-start-monitor)

    ;; 終了処理
    (set-process-sentinel
     my/org-latex-process
     (lambda (proc _event)
       (when (memq (process-status proc) '(exit signal))
         (setq my/org-latex-process nil)
         (if (= (process-exit-status proc) 0)
             (message "✔ PDF生成完了（%ds）"
                      my/org-latex-export-timer)
           (progn
             (message "✘ PDF生成失敗（%ds）— *org-latexmk* を表示"
                      my/org-latex-export-timer)
             (when-let ((buf (get-buffer "*org-latexmk*")))
               (display-buffer
                buf
                '((display-buffer-at-bottom)
                  (window-height . 0.3)))
               (with-current-buffer buf
                 (goto-char (point-max)))))))))))
#+end_src

(defun my/org-latex-export-stop ()
  "Stop running latexmk process politely, then force if needed."
  (interactive)
  (cond
   ((and my/org-latex-process (process-live-p my/org-latex-process))
    ;; まずは Ctrl-C 相当

(defun my/plist-get* (plist key &optional default)
  "Get KEY from PLIST, or DEFAULT if missing."

  "Render MDI icon HTML (or empty string if ICON is nil)."

(defun my/osbe--insert-my-special-block ()
  (interactive)
  (let* ((types '(("note" . "Note block")
                  ("note-fold" . "Foldable note block")
                  ("tip" . "Tip block")
                  ("tip-fold" . "Foldable tip block")
                  ("warning" . "Warning block")
                  ("warning-fold" . "Foldable warning block")
                  ("important" . "Important block")
                  ("important-fold" . "Foldable important block")
                  ("caution" . "Caution block")
                  ("caution-fold" . "Foldable caution block")))
         (type (completing-read "Block type: " types nil t))
         (title (read-string "Title: " (cdr (assoc type types)))))
    (insert (format "#+begin_%s %s\n\n#+end_%s" type title type))
    (forward-line -1)))

        (:around (orig-fun &rest args) handle-special-block)
      (if (string= (car args) "my-special-block")
          (my/osbe--insert-my-special-block)
